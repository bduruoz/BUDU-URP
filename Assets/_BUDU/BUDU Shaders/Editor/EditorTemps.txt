    // feature blok

    
        #region !

        EditorGUILayout.BeginVertical(style);
        if (check!)

            EditorGUILayout.BeginVertical();

    // ----------- SHADER DEFAULTS + ABOUT
    
        #region Shader Defaults
        style.normal.background = MakeBackground(1, 1, bdColors.GrayP(18, 204));
        style.fontSize = 16;
        style.normal.textColor = bdColors.NexusOrange();
        EditorGUILayout.BeginVertical(style);
        checkDef = EditorGUILayout.ToggleLeft("SHADER DEFAULTS", checkDef, style);
        targetMat.SetInt("_CheckDef", Convert.ToInt16(checkDef));
        EditorGUILayout.EndVertical();
        style.normal.background = MakeBackground(1, 1, bdColors.Transparent(0));
        EditorGUILayout.BeginVertical(style);
        if(checkDef)
        {
            EditorGUI.indentLevel++;
            EditorGUILayout.Space(2);

            #region Workflow Mode
            string[] workflowOptions = { "Metallic", "Specular" };
            int workflow = targetMat.GetFloat("_WorkflowMode") == 1 ? 1 : 0;
            workflow = EditorGUILayout.Popup("Workflow Mode", workflow, workflowOptions);
            targetMat.SetFloat("_WorkflowMode", workflow);
            if (workflow == 0)
            {
                targetMat.DisableKeyword("_SPECULAR_SETUP");
                MaterialProperty met = ShaderGUI.FindProperty("_Metallic", properties);
                materialEditor.ShaderProperty(met, "Metallic");
            }
            else
            {
                targetMat.EnableKeyword("_SPECULAR_SETUP");
                MaterialProperty spcCol = ShaderGUI.FindProperty("_Specular_Color", properties);
                materialEditor.ShaderProperty(spcCol, "Specular Color");
            }

            MaterialProperty smt = ShaderGUI.FindProperty("_Smoothness", properties);
            materialEditor.ShaderProperty(smt, "Smoothness");
            #endregion

            #region Surface Type
            string[] surfaceOptions = { "Opaque", "Transparent" };
            int surfaceType = targetMat.GetFloat("_Surface") == 1 ? 1 : 0;
            surfaceType = EditorGUILayout.Popup("Surface Type", surfaceType, surfaceOptions);
            targetMat.SetFloat("_Surface", surfaceType);

            if (surfaceType == 0)
            {
                // Opaque
                targetMat.SetOverrideTag("RenderType", "Opaque");
                targetMat.DisableKeyword("_SURFACE_TYPE_TRANSPARENT");
                targetMat.SetInt("_ZTest", (int)UnityEngine.Rendering.CompareFunction.LessEqual);
            }
            else
            {
                // Transparent
                targetMat.SetOverrideTag("RenderType", "Transparent");
                targetMat.SetOverrideTag("Queue", "Transparent");
                targetMat.EnableKeyword("_SURFACE_TYPE_TRANSPARENT");

                string[] depthOptions = { "Auto", "Force Enabled", "Force Disabled" };
                int current = 0; // Auto
                float zwrite = targetMat.GetFloat("_ZWrite");
                if (zwrite == 1) current = 1; // Force Enabled
                else if (zwrite == 0) current = 2; // Disabled

                int selected = EditorGUILayout.Popup("Depth Write", current, depthOptions);

                // Seçime göre sadece Force Enabled / Disabled uygula, Auto bırak shader'a
                switch (selected)
                {
                    case 0: break; // Auto, hiçbir şey yapma
                    case 1: targetMat.SetFloat("_ZWrite", 1); break;
                    case 2: targetMat.SetFloat("_ZWrite", 0); break;
                }

                // Depth Test
                string[] depthTestOptions = { "Disabled", "Never", "Less", "Equal", "LEqual", "Greater", "NotEqual", "GEqual", "Always" };
                int depthTest = (int)targetMat.GetFloat("_ZTest");
                depthTest = EditorGUILayout.Popup("Depth Test", depthTest, depthTestOptions);
                targetMat.SetFloat("_ZTest", depthTest);
            }
            #endregion

            #region Blending Modes
            // Dropdown Blending Mode
            MaterialProperty bmode = ShaderGUI.FindProperty("_BLENDING_MODE", properties);
            materialEditor.ShaderProperty(bmode, "Blending Mode");
            blendingMode = targetMat.GetInt("_BLENDING_MODE");

            switch (blendingMode)
            {
                case 0:
                    targetMat.SetFloat("_SrcBlend", (float)UnityEngine.Rendering.BlendMode.SrcAlpha);
                    targetMat.SetFloat("_DstBlend", (float)UnityEngine.Rendering.BlendMode.OneMinusSrcAlpha);
                    break;
                case 1:
                    targetMat.SetFloat("_SrcBlend", (float)UnityEngine.Rendering.BlendMode.One);
                    targetMat.SetFloat("_DstBlend", (float)UnityEngine.Rendering.BlendMode.OneMinusSrcAlpha);
                    break;
                case 2:
                    targetMat.SetFloat("_SrcBlend", (float)UnityEngine.Rendering.BlendMode.One);
                    targetMat.SetFloat("_DstBlend", (float)UnityEngine.Rendering.BlendMode.One);
                    break;
                case 3:
                    targetMat.SetFloat("_SrcBlend", (float)UnityEngine.Rendering.BlendMode.DstColor);
                    targetMat.SetFloat("_DstBlend", (float)UnityEngine.Rendering.BlendMode.Zero);
                    break;
            }
            #endregion

            #region Render Face
            string[] faceOptions = { "Both", "Back", "Front" };
            int face = (int)targetMat.GetFloat("_Cull");
            face = EditorGUILayout.Popup("Render Face", face, faceOptions);
            targetMat.SetFloat("_Cull", face);
            #endregion

            #region Cast Shadows
            MaterialProperty castShadows = ShaderGUI.FindProperty("_Cast_Shadows", properties);
            materialEditor.ShaderProperty(castShadows, "Cast Shadows");
            targetMat.SetInt("_CastShadow", Convert.ToInt16((int)castShadows.floatValue));
            targetMat.SetShaderPassEnabled("SHADOWCASTER", Convert.ToInt16((int)castShadows.floatValue) == 1 ? true : false);
            #endregion

            #region Receive Shadows
            MaterialProperty receiveShadow = ShaderGUI.FindProperty("_Receive_Shadows", properties);
            materialEditor.ShaderProperty(receiveShadow, "Receive Shadows");
            targetMat.SetInt("_ReceiveShadow", Convert.ToInt16((int)receiveShadow.floatValue));
            if (receiveShadow.floatValue < 1f)
            {
                targetMat.EnableKeyword(keyword: "_RECEIVE_SHADOWS_OFF");
            }
            else
            {
                targetMat.DisableKeyword(keyword: "_RECEIVE_SHADOWS_OFF");
            }
            #endregion

            materialEditor.RenderQueueField();
            materialEditor.EnableInstancingField();
            materialEditor.DoubleSidedGIField();
            EditorGUI.indentLevel--;
        }
        EditorGUILayout.EndVertical();
        EditorGUILayout.Space(1);
        #endregion

        #region BUDU Copyright
        EditorGUILayout.Space(5);
        style.normal.textColor = bdColors.NexusOrange(153);
        style.fontSize = 10;
        style.fontStyle = FontStyle.Italic;

        aboutFold = EditorGUILayout.Foldout(aboutFold, "About", toggleOnLabelClick: true, style);
        if(aboutFold)
        {
            EditorGUILayout.BeginHorizontal();
            GUILayout.FlexibleSpace();
            Texture budulogo = (Texture)AssetDatabase.LoadAssetAtPath("Assets/_BUDU/GUI/BUDUGAMES-About.png", typeof(Texture));
            Rect rect = GUILayoutUtility.GetRect(400,50,GUILayout.ExpandWidth(false));
            GUI.DrawTexture(rect, budulogo, ScaleMode.ScaleToFit, true);
            GUILayout.FlexibleSpace();
            EditorGUILayout.EndHorizontal();
        }
        #endregion







//////////////////////////////////////////////////////////////////////////////////////////////////////



    // ------------ SETTINGS SETTING
        
        #region /// Settings
        style.normal.background = MakeBackground(1, 32, bdColors.GrayP(18,204));
        style.fontSize = 16;
        style.normal.textColor = bdColors.NexusOrange();

        EditorGUILayout.BeginVertical(style);
        xxxDef = EditorGUILayout.ToggleLeft("--- SETTINGS", xxxDef, style);
        targetMat.SetInt("_xxxDef", Convert.ToInt16(xxxDef));
        EditorGUILayout.EndVertical();

        style.normal.background = MakeBackground(1, 1, bdColors.Transparent(0));
        EditorGUILayout.BeginVertical(style);
        if(xxxDef)
        {
            EditorGUI.indentLevel++;
            EditorGUILayout.Space(2);

            ///

            EditorGUILayout.Space(1);
            EditorGUI.indentLevel--;
        }
        EditorGUILayout.EndVertical();
        EditorGUILayout.Space(1);
        #endregion
